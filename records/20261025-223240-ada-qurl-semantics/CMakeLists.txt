cmake_minimum_required(VERSION 3.24)
project(ada_qurl_semantics LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ADA_TESTING OFF CACHE BOOL "" FORCE)
set(ADA_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(ADA_TOOLS OFF CACHE BOOL "" FORCE)
set(ADA_INCLUDE_URL_PATTERN OFF CACHE BOOL "" FORCE)
set(ADA_USE_SIMDUTF OFF CACHE BOOL "" FORCE)
set(ADA_BUILD_SINGLE_HEADER_LIB OFF CACHE BOOL "" FORCE)
set(CMAKE_DISABLE_FIND_PACKAGE_Python3 ON)
find_package(Qt5 5.15 REQUIRED COMPONENTS Core)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../3rdparty/ada" "${CMAKE_CURRENT_BINARY_DIR}/ada")
add_executable(ada_qurl_probe probe.cpp)
target_link_libraries(ada_qurl_probe PRIVATE Qt5::Core ada::ada)
